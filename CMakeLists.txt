cmake_minimum_required(VERSION 3.20)
project(murmur VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core DBus Network Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.65)
add_library(murmur-dsp STATIC src/dsp.cpp)
target_include_directories(murmur-dsp PUBLIC src)
add_library(murmur-model STATIC src/model.cpp)
target_link_libraries(murmur-model PUBLIC Qt6::Core murmur-dsp)
add_executable(murmur-daemon src/daemon_main.cpp src/daemon.cpp src/daemon.hpp src/engine.cpp src/engine.hpp src/background_portal.cpp src/background_portal.hpp)
target_link_libraries(murmur-daemon PRIVATE Qt6::Core Qt6::DBus Qt6::Network murmur-model PkgConfig::PIPEWIRE)
add_executable(murmur src/main.cpp src/window.cpp src/window.hpp src/client.cpp src/client.hpp)
target_link_libraries(murmur PRIVATE Qt6::Widgets Qt6::Network murmur-model)
foreach(target murmur murmur-daemon murmur-model murmur-dsp)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wconversion)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(dsp-tests tests/dsp_tests.cpp)
  target_link_libraries(dsp-tests PRIVATE murmur-dsp)
  add_test(NAME dsp COMMAND dsp-tests)
  add_executable(model-tests tests/model_tests.cpp)
  target_link_libraries(model-tests PRIVATE murmur-model)
  add_test(NAME model COMMAND model-tests)
  add_executable(ui-tests tests/ui_tests.cpp src/window.cpp src/window.hpp src/client.cpp src/client.hpp)
  target_include_directories(ui-tests PRIVATE src)
  target_link_libraries(ui-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test murmur-model)
  add_test(NAME ui COMMAND ui-tests)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
  add_executable(background-portal-tests tests/background_portal_tests.cpp src/background_portal.cpp src/background_portal.hpp)
  target_include_directories(background-portal-tests PRIVATE src)
  target_link_libraries(background-portal-tests PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
  add_test(NAME background-portal COMMAND ${DBUS_RUN_SESSION} -- ${CMAKE_COMMAND} -E env MURMUR_PRIVATE_PORTAL_TEST=1 $<TARGET_FILE:background-portal-tests>)
  add_executable(tone-source tests/tone_source.cpp)
  target_link_libraries(tone-source PRIVATE PkgConfig::PIPEWIRE)
endif()
include(GNUInstallDirs)
configure_file(packaging/murmur.service.in murmur.service @ONLY)
option(MURMUR_FLATPAK "Install Flatpak AppStream metadata instead of the native user service" OFF)
install(TARGETS murmur murmur-daemon RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(MURMUR_FLATPAK)
  install(FILES packaging/io.github.murmur_voice.Murmur.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
else()
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/murmur.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/user)
endif()
install(FILES packaging/murmur.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/murmur.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/murmur)
